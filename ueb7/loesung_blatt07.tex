%% ============================================================
\subsection*{Aufgabe 26}
%% ============================================================

\textbf{Angabe.} Berechne die Determinanten effizient (per Hand, ohne Rechner):
\[
A = \begin{pmatrix} 234 & 235 & 236 \\ 235 & 236 & 237 \\ 236 & 237 & 239 \end{pmatrix},\qquad
B = \begin{pmatrix} 21 & 23 & 0 & 27 & 2 \\ 0 & 2 & 0 & 7 & 0 \\ 11 & 12 & 13 & 14 & 15 \\ 1 & 41 & 0 & 43 & 0 \\ 0 & 1 & 0 & 4 & 0 \end{pmatrix}.
\]

\begin{begriffe}
\begin{itemize}[leftmargin=*, nosep]
    \item \textbf{Determinante} -- eine Zahl, die einer quadratischen Matrix zugeordnet wird. $\det(A) = 0 \iff$ Zeilen (oder Spalten) sind linear abh\"angig.
    \item \textbf{Eigenschaft 1: Zeilenumformungen.} $Z_i \to Z_i + \lambda Z_j$ \"andert die Determinante \textbf{nicht}.
    \item \textbf{Eigenschaft 2: Zeilentausch.} \"Andert das Vorzeichen der Determinante.
    \item \textbf{Eigenschaft 3: Spaltenumformungen.} Genau wie bei Zeilen.
    \item \textbf{Laplace-Entwicklung} -- Verfahren, um eine gro\ss e Determinante in mehrere kleinere zu zerlegen. Anschaulich: ich w\"ahle eine Zeile (oder Spalte), gehe jeden Eintrag dieser Zeile durch und reduziere damit die $n \times n$-Determinante auf $n$ kleinere $(n-1) \times (n-1)$-Determinanten. Genaue Anleitung weiter unten beim ersten Einsatz.
    \item \textbf{Trick}: Zeilen/Spalten mit vielen Nullen suchen $\Rightarrow$ in der Summe werden viele Terme automatisch $0$, und die Rechnung wird viel k\"urzer.
\end{itemize}
\end{begriffe}

\begin{wastun}
\begin{enumerate}[leftmargin=*, nosep]
    \item \textbf{Matrix $A$:} Zahlen sehen wie aufeinanderfolgende Werte aus. Mit Zeilenumformungen $Z_2 \to Z_2 - Z_1$, $Z_3 \to Z_3 - Z_1$ entstehen kleine Zahlen.
    \item \textbf{Matrix $B$:} Die Spalte 3 hat fast nur Nullen (nur ein Eintrag $\neq 0$). Laplace-Entwicklung nach Spalte 3 reduziert auf eine $4 \times 4$-Matrix. Die wieder hat eine Spalte mit fast nur Nullen $\to$ weiter entwickeln, bis nur noch $2 \times 2$ \"ubrig.
\end{enumerate}
\end{wastun}

%% --- 26 A ---
\subsubsection*{Determinante von $A$}

\textbf{Schritt 1: Zeilenumformungen, um die gro\ss en Zahlen loszuwerden.}

Beobachtung: Die Zeilen unterscheiden sich nur um wenig. $Z_2 - Z_1$ und $Z_3 - Z_1$ heben die ``$234$er-Werte'' weg, weil aufeinanderfolgende Zahlen sich nur um $1$ unterscheiden.

\textbf{(a) $Z_2 \to Z_2 - Z_1$ und $Z_3 \to Z_3 - Z_1$} (Determinante \"andert sich nicht bei dieser Umformung):
\begin{itemize}[leftmargin=*, nosep]
    \item neue $Z_2 = (235-234,\; 236-235,\; 237-236) = (1, 1, 1)$
    \item neue $Z_3 = (236-234,\; 237-235,\; 239-236) = (2, 2, 3)$
\end{itemize}
\[
\det A = \det \begin{pmatrix} 234 & 235 & 236 \\ 1 & 1 & 1 \\ 2 & 2 & 3 \end{pmatrix}
\]

\textbf{(b) Warum jetzt $Z_3 \to Z_3 - 2Z_2$?} Ich will Nullen in Zeile 3 erzeugen, damit die sp\"atere Laplace-Entwicklung trivial wird. $Z_3$ hat $(2, 2, 3)$ und $Z_2 = (1, 1, 1)$. $Z_3 - 2 \cdot Z_2$ macht aus den ersten zwei Eintr\"agen $0$:
\begin{itemize}[leftmargin=*, nosep]
    \item neue $Z_3 = (2 - 2 \cdot 1,\; 2 - 2 \cdot 1,\; 3 - 2 \cdot 1) = (0, 0, 1)$
\end{itemize}
\[
= \det \begin{pmatrix} 234 & 235 & 236 \\ 1 & 1 & 1 \\ 0 & 0 & 1 \end{pmatrix}
\]

\textbf{Schritt 2: Laplace-Entwicklung nach Zeile 3.}

Zeile 3 ist $(0, 0, 1)$. Die zwei Nullen liefern Beitrag $0$, also bleibt nur Eintrag $(3,3) = 1$. Streiche \textbf{Zeile 3 und Spalte 3} weg (grau):

\[
\begin{pmatrix}
234 & 235 & \color{gray}236 \\
1 & 1 & \color{gray}1 \\
\color{gray}0 & \color{gray}0 & \color{gray}1
\end{pmatrix}
\;\longrightarrow\;
\det = 1 \cdot \underbrace{(-1)^{3+3}}_{+1} \cdot \det \begin{pmatrix} 234 & 235 \\ 1 & 1 \end{pmatrix}
= 234 - 235 = -1.
\]

\[
\fbox{\parbox{0.93\textwidth}{\centering
$\det A = -1$.
}}
\]

%% --- 26 B ---
\subsubsection*{Determinante von $B$}

\textbf{Idee:} Spalten/Zeilen mit fast nur Nullen finden und wegklappen, bis nur noch $2 \times 2$ \"ubrig.

\textbf{Warum Nullen suchen?} Bei der Laplace-Entwicklung wird jeder Eintrag der Zeile/Spalte mit seinem Minor multipliziert. Ist der Eintrag $0$, ist sein ganzer Term $0 \cdot \text{Minor} = 0$ -- der Minor muss gar nicht erst berechnet werden. Eine Spalte mit \textbf{einer einzigen} Zahl $\neq 0$ reduziert die Arbeit auf einen einzigen $(n-1)\times(n-1)$-Minor.

\textbf{Schritt 1: $5\times 5 \to 4\times 4$. Entwicklung nach Spalte 3.}

\textbf{Warum Spalte 3?} Schau dir die Spalten von $B$ an: Spalte 3 hat den Inhalt $(0, 0, 13, 0, 0)$ -- vier von f\"unf Eintr\"agen sind $0$. Keine andere Spalte (und keine Zeile) ist so leer. Deshalb entwickeln wir genau hier.

\textbf{Was bleibt \"ubrig?} Nur Eintrag $(3,3) = 13$ liefert einen nichttrivialen Beitrag. Streiche Zeile 3 und Spalte 3:

\[
\begin{pmatrix}
21 & 23 & \color{gray}0 & 27 & 2 \\
0 & 2 & \color{gray}0 & 7 & 0 \\
\color{gray}11 & \color{gray}12 & \color{gray}13 & \color{gray}14 & \color{gray}15 \\
1 & 41 & \color{gray}0 & 43 & 0 \\
0 & 1 & \color{gray}0 & 4 & 0
\end{pmatrix}
\;\longrightarrow\;
\det B = 13 \cdot \underbrace{(-1)^{3+3}}_{+1} \cdot \det \underbrace{\begin{pmatrix} 21 & 23 & 27 & 2 \\ 0 & 2 & 7 & 0 \\ 1 & 41 & 43 & 0 \\ 0 & 1 & 4 & 0 \end{pmatrix}}_{=: M}
\]

\textbf{Schritt 2: $4\times 4 \to 3\times 3$. Entwicklung von $M$ nach Spalte 4.}

\textbf{Warum Spalte 4 von $M$?} Wieder Spalten von $M$ pr\"ufen: Spalte 4 lautet $(2, 0, 0, 0)$ -- drei Nullen, nur ein Eintrag $\neq 0$. Wieder die ``leerste'' Spalte. Streiche Zeile 1 und Spalte 4:

\[
\begin{pmatrix}
\color{gray}21 & \color{gray}23 & \color{gray}27 & \color{gray}2 \\
0 & 2 & 7 & \color{gray}0 \\
1 & 41 & 43 & \color{gray}0 \\
0 & 1 & 4 & \color{gray}0
\end{pmatrix}
\;\longrightarrow\;
\det M = 2 \cdot \underbrace{(-1)^{1+4}}_{-1} \cdot \det \underbrace{\begin{pmatrix} 0 & 2 & 7 \\ 1 & 41 & 43 \\ 0 & 1 & 4 \end{pmatrix}}_{=: N}
= -2 \cdot \det N.
\]

\textbf{Schritt 3: $3\times 3 \to 2\times 2$. Entwicklung von $N$ nach Spalte 1.}

\textbf{Warum Spalte 1 von $N$?} Spalte 1 lautet $(0, 1, 0)$ -- zwei Nullen, nur ein Eintrag $\neq 0$. Streiche Zeile 2 und Spalte 1:

\[
\begin{pmatrix}
\color{gray}0 & 2 & 7 \\
\color{gray}1 & \color{gray}41 & \color{gray}43 \\
\color{gray}0 & 1 & 4
\end{pmatrix}
\;\longrightarrow\;
\det N = 1 \cdot \underbrace{(-1)^{2+1}}_{-1} \cdot \det \begin{pmatrix} 2 & 7 \\ 1 & 4 \end{pmatrix}
= -1 \cdot (8 - 7) = -1.
\]

\textbf{Schritt 4: Zur\"uckrechnen.}

\[
\det M = -2 \cdot (-1) = 2, \qquad \det B = 13 \cdot 2 = 26.
\]

\[
\fbox{\parbox{0.93\textwidth}{\centering
$\det B = 26$.
}}
\]

\subsection*{Aufgabe 27}
%% ============================================================

\textbf{Angabe.}
\[
A = \begin{pmatrix} a & 1 & 2 & 3 \\ 1 & 2 & 3 & 2 \\ 2 & 3 & 2 & 1 \\ 3 & 2 & 1 & b \end{pmatrix} \in \mathbb{R}^{4\times 4},\quad a, b \in \mathbb{R}.
\]
(a) $\det A$ berechnen. \quad (b) Bedingung an $a, b$, sodass $Ax = 0$ unendlich viele L\"osungen hat.

\begin{begriffe}
\begin{itemize}[leftmargin=*, nosep]
    \item \textbf{Parameter $a, b$} -- unbestimmte Zahlen, die im Ergebnis stehen bleiben d\"urfen. $\det A$ wird ein Ausdruck in $a$ und $b$.
    \item \textbf{$Ax = 0$ hat unendlich viele L\"osungen} $\iff$ $A$ ist nicht invertierbar $\iff$ $\det A = 0$. (Das homogene System hat \emph{immer} die triviale L\"osung $x = 0$; weitere L\"osungen existieren genau dann, wenn $\det A = 0$.)
\end{itemize}
\end{begriffe}

\begin{wastun}
\begin{enumerate}[leftmargin=*, nosep]
    \item \textbf{(a)} Determinante per Laplace-Entwicklung nach der ersten Zeile berechnen (4 Minoren, jeder $3\times 3$). Parameter $a, b$ bleiben mitgeschleppt.
    \item \textbf{(b)} $\det A = 0$ setzen und nach $a, b$ aufl\"osen.
\end{enumerate}
\end{wastun}

%% --- 27(a) ---
\subsubsection*{(a) Determinante von $A$}

\textbf{Schritt 1: Laplace-Entwicklung nach der ersten Zeile.}

Notation: $a_{ij}$ steht f\"ur den Eintrag in \textbf{Zeile $i$, Spalte $j$}. Hier:
\[
A = \begin{pmatrix}
\underbrace{a}_{a_{11}} & \underbrace{1}_{a_{12}} & \underbrace{2}_{a_{13}} & \underbrace{3}_{a_{14}} \\
1 & 2 & 3 & 2 \\
2 & 3 & 2 & 1 \\
3 & 2 & 1 & b
\end{pmatrix}
\]
also $a_{11} = a$, $a_{12} = 1$, $a_{13} = 2$, $a_{14} = 3$.

Vorzeichen f\"ur Zeile 1 (Schachbrett): $+\;-\;+\;-$. Damit:
\[
\det A = a \cdot M_{11} - 1 \cdot M_{12} + 2 \cdot M_{13} - 3 \cdot M_{14}.
\]
$M_{1j}$: Zeile 1 und Spalte $j$ wegstreichen.

\textbf{Schritt 2: Vier Minoren ausrechnen.}

Pro Minor $M_{1j}$ wird in $A$ \textbf{Zeile 1 und Spalte $j$} weggestrichen (grau dargestellt) -- die schwarzen Eintr\"age bilden die $3\times 3$-Matrix.

\textbf{(a) $M_{11}$ -- Zeile 1 und Spalte 1 weg.}
\[
\begin{pmatrix}
\color{gray}a & \color{gray}1 & \color{gray}2 & \color{gray}3 \\
\color{gray}1 & 2 & 3 & 2 \\
\color{gray}2 & 3 & 2 & 1 \\
\color{gray}3 & 2 & 1 & b
\end{pmatrix}
\;\longrightarrow\;
M_{11} = \det \begin{pmatrix} 2 & 3 & 2 \\ 3 & 2 & 1 \\ 2 & 1 & b \end{pmatrix}
= 2(2b - 1) - 3(3b - 2) + 2(3 - 4)
= -5b + 2.
\]

\textbf{(b) $M_{12}$ -- Zeile 1 und Spalte 2 weg.}
\[
\begin{pmatrix}
\color{gray}a & \color{gray}1 & \color{gray}2 & \color{gray}3 \\
1 & \color{gray}2 & 3 & 2 \\
2 & \color{gray}3 & 2 & 1 \\
3 & \color{gray}2 & 1 & b
\end{pmatrix}
\;\longrightarrow\;
M_{12} = \det \begin{pmatrix} 1 & 3 & 2 \\ 2 & 2 & 1 \\ 3 & 1 & b \end{pmatrix}
= 1(2b - 1) - 3(2b - 3) + 2(2 - 6)
= -4b.
\]

\textbf{(c) $M_{13}$ -- Zeile 1 und Spalte 3 weg.}
\[
\begin{pmatrix}
\color{gray}a & \color{gray}1 & \color{gray}2 & \color{gray}3 \\
1 & 2 & \color{gray}3 & 2 \\
2 & 3 & \color{gray}2 & 1 \\
3 & 2 & \color{gray}1 & b
\end{pmatrix}
\;\longrightarrow\;
M_{13} = \det \begin{pmatrix} 1 & 2 & 2 \\ 2 & 3 & 1 \\ 3 & 2 & b \end{pmatrix}
= 1(3b - 2) - 2(2b - 3) + 2(4 - 9)
= -b - 6.
\]

\textbf{(d) $M_{14}$ -- Zeile 1 und Spalte 4 weg.}
\[
\begin{pmatrix}
\color{gray}a & \color{gray}1 & \color{gray}2 & \color{gray}3 \\
1 & 2 & 3 & \color{gray}2 \\
2 & 3 & 2 & \color{gray}1 \\
3 & 2 & 1 & \color{gray}b
\end{pmatrix}
\;\longrightarrow\;
M_{14} = \det \begin{pmatrix} 1 & 2 & 3 \\ 2 & 3 & 2 \\ 3 & 2 & 1 \end{pmatrix}
= 1(3 - 4) - 2(2 - 6) + 3(4 - 9)
= -8.
\]

\textbf{Schritt 3: Zusammensetzen.}

\begin{align*}
\det A &= a \cdot (-5b+2) - 1 \cdot (-4b) + 2 \cdot (-b - 6) - 3 \cdot (-8) \\
&= -5ab + 2a + 4b - 2b - 12 + 24 \\
&= -5ab + 2a + 2b + 12.
\end{align*}

\[
\fbox{\parbox{0.93\textwidth}{\centering
$\det A = -5ab + 2a + 2b + 12$.
}}
\]

%% --- 27(b) ---
\subsubsection*{(b) Bedingung f\"ur unendlich viele L\"osungen}

\textbf{Schritt 1: $\det A = 0$ ansetzen.}

$Ax = 0$ hat unendlich viele L\"osungen $\iff \det A = 0$:
\[
-5ab + 2a + 2b + 12 = 0 \quad\iff\quad 5ab - 2a - 2b - 12 = 0.
\]

\textbf{Schritt 2: Faktorisieren.}

\textbf{Warum \"uberhaupt faktorisieren?} Die Gleichung $5ab - 2a - 2b - 12 = 0$ steht in einer ``unsch\"onen'' Form -- man sieht $a$, $b$ und das Produkt $ab$ nebeneinander. Eine Form $(\dots)(\dots) = \text{Zahl}$ ist deutlich angenehmer: man hat dann zwei klare Faktoren, kann konkrete L\"osungspaare leichter bauen und sieht direkt, dass es \emph{unendlich} viele $(a, b)$ gibt.

\textbf{Warum mit $5$ multiplizieren?} Wir wollen die Gleichung in die Form $(5a + p)(5b + q) = \text{Zahl}$ bringen, weil das den Faktor $5ab$ aus der Originalgleichung mit dem Faktor $25ab = (5a)(5b)$ aus dem Produkt verbindet. Genauer:
\[
(5a + p)(5b + q) = 25ab + 5qa + 5pb + pq.
\]
Damit Koeffizientenvergleich m\"oglich wird, brauchen wir auf der linken Seite den Vorfaktor $25$ vor $ab$. Daher multiplizieren wir die ganze Gleichung mit $5$ -- das ist der Faktor, der $5ab$ zu $25ab$ macht, ohne neue Variablen einzuf\"uhren.

\textbf{Schritt 2a: Mit $5$ multiplizieren.}
\[
5 \cdot (5ab - 2a - 2b - 12) = 0 \;\Longleftrightarrow\; 25ab - 10a - 10b - 60 = 0.
\]

\textbf{Schritt 2b: Form $(5a + p)(5b + q)$ raten.}

Wir wollen die linke Seite als Produkt schreiben. Versuchsansatz: \emph{irgendwas mit $5a$} mal \emph{irgendwas mit $5b$}, also
\[
(5a + p)(5b + q),
\]
wobei $p$ und $q$ \textbf{noch unbekannte Zahlen} sind, die wir suchen ($p, q$ sind nur Platzhalter -- wie $x$ und $y$ in einer Gleichung).

\textbf{Schritt 2b.1: Klammern ausmultiplizieren} (FOIL):
\[
(5a + p)(5b + q) = \underbrace{5a \cdot 5b}_{= 25ab} + \underbrace{5a \cdot q}_{= 5qa} + \underbrace{p \cdot 5b}_{= 5pb} + \underbrace{p \cdot q}_{= pq}
= 25ab + 5q\, a + 5p\, b + pq.
\]

\textbf{Schritt 2b.2: Mit unserer Gleichung vergleichen.} Wir m\"ochten, dass das gleich $25ab - 10a - 10b + (\text{Zahl})$ wird. Koeffizientenvergleich:
\begin{itemize}[leftmargin=*, nosep]
    \item Koeffizient von $a$: $5q$ soll gleich $-10$ sein $\Rightarrow q = -2$.
    \item Koeffizient von $b$: $5p$ soll gleich $-10$ sein $\Rightarrow p = -2$.
\end{itemize}

\textbf{Schritt 2b.3: $p = q = -2$ einsetzen.}
\[
(5a - 2)(5b - 2) = 25ab - 10a - 10b + \underbrace{(-2)(-2)}_{= 4} = 25ab - 10a - 10b + 4.
\]

\textbf{Schritt 2c: Konstanten ausgleichen.} In unserer Gleichung steht $-60$ statt $+4$. Differenz: $4 - (-60) = 64$. Genau diese $64$ landet auf der rechten Seite:
\[
25ab - 10a - 10b - 60 = 0
\;\Longleftrightarrow\; \underbrace{(25ab - 10a - 10b + 4)}_{= (5a-2)(5b-2)} - 64 = 0
\;\Longleftrightarrow\; (5a-2)(5b-2) = 64.
\]

\[
\fbox{\parbox{0.93\textwidth}{\centering
$Ax = 0$ hat unendlich viele L\"osungen $\iff$ $(5a-2)(5b-2) = 64$.\\[2pt]
\"Aquivalent: $5ab - 2a - 2b = 12$.
}}
\]

\subsection*{Aufgabe 28}
%% ============================================================

\textbf{Angabe.} Sei $n \geq 3$ ungerade, $A \in \mathbb{R}^{n \times n}$ schiefsymmetrisch ($A^\top = -A$). Zeige: die Zeilen von $A$ sind linear abh\"angig.

\begin{begriffe}
\begin{itemize}[leftmargin=*, nosep]
    \item \textbf{Schiefsymmetrisch}: $A^\top = -A$, d.\,h.\ $a_{ij} = -a_{ji}$.
    \item Zeilen sind \textbf{linear abh\"angig} $\iff$ $\det A = 0$.
    \item Es gen\"ugt also, $\det A = 0$ zu zeigen.
\end{itemize}
\end{begriffe}

\begin{wastun}
\begin{enumerate}[leftmargin=*, nosep]
    \item Mit Determinanten-Eigenschaften zeigen, dass $\det A = -\det A$ folgt.
    \item Daraus $\det A = 0$, daher Zeilen linear abh\"angig.
\end{enumerate}
\end{wastun}

\textbf{Strategie.} Wir wollen ``Zeilen linear abh\"angig'' zeigen. Wir wissen:
$$\text{Zeilen linear abh\"angig} \iff \det(A) = 0.$$
Es reicht also zu zeigen, dass $\det(A) = 0$. Daf\"ur basteln wir aus zwei Determinanten-Eigenschaften die Gleichung $\det(A) = -\det(A)$ -- daraus folgt sofort $\det(A) = 0$.

\medskip

\textbf{Schritt 1: Zwei Determinanten-Eigenschaften aus der Vorlesung sammeln.}

\textbf{(a) Transponieren \"andert die Determinante nicht.} F\"ur jede quadratische Matrix gilt:
\[
\det(A^\top) = \det(A).
\]

\textbf{(b) Skalar mal Matrix.} F\"ur $\lambda \in \mathbb{R}$ und $A \in \mathbb{R}^{n \times n}$:
\[
\det(\lambda A) = \lambda^n \det(A).
\]
\emph{Warum $\lambda^n$?} Bei $\lambda A$ wird jede der $n$ Zeilen mit $\lambda$ multipliziert. Pro skalierter Zeile zieht ein $\lambda$ aus der Determinante heraus -- bei $n$ Zeilen also $\lambda^n$ insgesamt.

Speziell f\"ur $\lambda = -1$:
\[
\det(-A) = (-1)^n \det(A).
\]

\medskip

\textbf{Schritt 2: Schiefsymmetrie ausnutzen.}

Die Voraussetzung $A^\top = -A$ bedeutet: die Matrizen $A^\top$ und $-A$ sind \textbf{dieselbe Matrix}. Also haben sie auch \textbf{dieselbe Determinante}:
\[
\det(A^\top) = \det(-A).
\]

\textbf{Schritt 3: Beide Eigenschaften in die Gleichung aus Schritt 2 einsetzen.}

Ausgangsgleichung (Schritt 2):
\[
\det(A^\top) = \det(-A).
\]

\textbf{(a) Linke Seite umschreiben.} Eigenschaft (a) sagt $\det(A^\top) = \det(A)$. Linke Seite ersetzen:
\[
\det(A) = \det(-A).
\]

\textbf{(b) Rechte Seite umschreiben.} Eigenschaft (b) sagt $\det(-A) = (-1)^n \det(A)$. Rechte Seite ersetzen:
\[
\det(A) = (-1)^n \det(A).
\]

\medskip

\textbf{Schritt 4: $n$ ungerade ausnutzen.}

F\"ur ungerades $n$ (z.\,B.\ $n = 3, 5, 7, \dots$) ist $(-1)^n = -1$. Damit wird:
\[
\det(A) = -\det(A).
\]

Beide $\det(A)$-Terme auf eine Seite bringen:
\[
\det(A) + \det(A) = 0 \;\Longleftrightarrow\; 2 \det(A) = 0 \;\Longleftrightarrow\; \det(A) = 0.
\]

\medskip

\textbf{Schritt 5: Folgerung ziehen.}

$\det(A) = 0 \Rightarrow$ die Zeilen von $A$ sind linear abh\"angig. $\square$

\medskip

\textbf{Zur Kontrolle: warum nur ungerades $n$?} F\"ur gerades $n$ ist $(-1)^n = +1$, dann lautet die Gleichung in Schritt 4 nur $\det(A) = \det(A)$ -- das ist keine neue Information, $\det A$ kann beliebig sein. Beispiel: $A = \begin{pmatrix} 0 & 1 \\ -1 & 0 \end{pmatrix}$ ist schiefsymmetrisch ($n=2$, gerade) mit $\det A = 1 \neq 0$. Das Argument funktioniert also wirklich nur f\"ur ungerades $n$.

\subsection*{Aufgabe 29}
%% ============================================================

\textbf{Angabe.} Sei $n \geq 3$, $A \in \mathbb{R}^{n \times n}$, jede Zeile von $A$ bildet eine arithmetische Folge. Zeige: $\det(A) = 0$.

\begin{begriffe}
\begin{itemize}[leftmargin=*, nosep]
    \item \textbf{Arithmetische Folge}: Jeder Schritt hat den gleichen Abstand. Z.\,B.\ $(2, 5, 8, 11, \dots)$ -- Differenz immer $+3$.
    \item Allgemein: $(a,\; a + d,\; a + 2d,\; a + 3d,\; \dots)$ mit Startwert $a$ und Schrittweite $d$.
    \item \textbf{Schl\"ussel}: F\"ur drei aufeinanderfolgende Glieder $x, y, z$ einer arithmetischen Folge gilt
    \[
    z - 2y + x = 0
    \]
    (denn $y - x = d = z - y$, also $z + x = 2y$). Diese ``$+1, -2, +1$''-Kombination ergibt immer Null.
\end{itemize}
\end{begriffe}

\subsubsection*{Hauptargument: Spaltenoperation gibt Nullspalte}

\textbf{Idee.} Wenn jede Zeile arithmetisch ist, gilt die Beziehung ``Spalte $1$ minus $2 \cdot$ Spalte $2$ plus Spalte $3$ $= 0$'' \textbf{Zeile f\"ur Zeile} -- jede Zeile hat ja drei aufeinanderfolgende Glieder einer arithmetischen Folge in den Spalten $1, 2, 3$.

\textbf{Konkretes Beispiel.} Schauen wir uns eine $3 \times 3$-Matrix an, deren Zeilen jeweils arithmetisch sind:
\[
A = \begin{pmatrix}
\color{red}2 & \color{red}5 & \color{red}8 \\
\color{blue}1 & \color{blue}3 & \color{blue}5 \\
\color{teal}10 & \color{teal}7 & \color{teal}4
\end{pmatrix}
\]
(Zeile 1: Schritt $+3$. Zeile 2: Schritt $+2$. Zeile 3: Schritt $-3$.)

Bilde $S_3 - 2 S_2 + S_1$ (Spalte 3 minus doppelte Spalte 2 plus Spalte 1) eintragsweise:
\begin{itemize}[leftmargin=*, nosep]
    \item Zeile 1: $\color{red}8 - 2\cdot \color{red}5 + \color{red}2 = 8 - 10 + 2 = 0$
    \item Zeile 2: $\color{blue}5 - 2\cdot \color{blue}3 + \color{blue}1 = 5 - 6 + 1 = 0$
    \item Zeile 3: $\color{teal}4 - 2\cdot \color{teal}7 + \color{teal}10 = 4 - 14 + 10 = 0$
\end{itemize}
Ergebnis: \textbf{Nullspalte}.

\textbf{Allgemein.} Zeile $i$ ist arithmetisch mit Startwert $a_i$ und Schrittweite $d_i$, also Spalten $(a_i,\; a_i + d_i,\; a_i + 2d_i,\; \dots)$. Eintrag von $S_3 - 2 S_2 + S_1$ in Zeile $i$:
\[
(a_i + 2d_i) - 2(a_i + d_i) + a_i = a_i(1 - 2 + 1) + d_i(2 - 2 + 0) = 0.
\]
Funktioniert f\"ur \emph{jede} Zeile $\Rightarrow$ neue Spalte 3 ist die Nullspalte.

\textbf{Folgerung.} Determinante \"andert sich bei Spaltenoperationen $S_j \to S_j + \lambda S_k$ nicht. Eine Matrix mit einer Nullspalte hat Determinante $0$. Daher:
\[
\fbox{$\det(A) = 0$.} \qquad \square
\]

\subsubsection*{Alternative Argumente (kurz)}

\begin{itemize}[leftmargin=*, nosep]
    \item \textbf{Lineare Abh\"angigkeit der Spalten}: $S_3 - 2S_2 + S_1 = 0$ ist eine nichttriviale Spaltenrelation $\Rightarrow$ Spalten linear abh\"angig $\Rightarrow \det A = 0$.
    \item \textbf{Nichttriviale L\"osung von $Ax = 0$}: W\"ahle $x = (1, -2, 1, 0, \dots, 0)^\top$. Dann ist $Ax = 0$ mit $x \neq 0$ $\Rightarrow \det A = 0$.
    \item \textbf{Spaltenraum-Dimension}: Jede Spalte $S_j$ ist von der Form $u + (j-1) v$ mit $u = (a_1, \dots, a_n)^\top$ und $v = (d_1, \dots, d_n)^\top$. Spaltenraum liegt in $\text{span}(u, v)$, hat also H\"ochstdimension $2$. F\"ur $n \geq 3$ folgt $\text{rang}(A) < n$, also $\det A = 0$.
\end{itemize}

\subsection*{Aufgabe 30}
%% ============================================================

\textbf{Angabe.}
\[
\begin{cases} 2x_1 + 5x_2 + 3x_3 = 5 \\ -x_1 - 4x_2 + x_3 = -3 \\ 2x_2 - 5x_3 = -1 \end{cases}
\]
auf zwei Wegen l\"osen: (a) \"uber die adjungierte Matrix, (b) Cramer'sche Regel.

\[
A = \begin{pmatrix} 2 & 5 & 3 \\ -1 & -4 & 1 \\ 0 & 2 & -5 \end{pmatrix},\quad
b = \begin{pmatrix} 5 \\ -3 \\ -1 \end{pmatrix}.
\]

\begin{begriffe}
\begin{itemize}[leftmargin=*, nosep]
    \item \textbf{Kofaktor} $C_{ij} = (-1)^{i+j} M_{ij}$, wobei $M_{ij}$ die Minor-Determinante (Matrix ohne Zeile $i$ und Spalte $j$) ist.
    \item \textbf{Kofaktormatrix} $\text{cof}(A) = (C_{ij})$ -- die Matrix aller Kofaktoren.
    \item \textbf{Adjungierte} (Adjunkte) $\text{adj}(A) = \text{cof}(A)^\top$ -- transponierte Kofaktormatrix.
    \item \textbf{Inverse \"uber Adjunkte}: $A^{-1} = \tfrac{1}{\det A} \cdot \text{adj}(A)$ (falls $\det A \neq 0$).
    \item \textbf{Cramer'sche Regel}: $x_i = \dfrac{\det A_i}{\det A}$, wobei $A_i$ aus $A$ entsteht, indem man die $i$-te Spalte durch $b$ ersetzt.
\end{itemize}
\end{begriffe}

\subsubsection*{Schritt 1: $\det A$ berechnen (f\"ur beide Methoden gebraucht)}

Laplace-Entwicklung nach erster Spalte. Eintr\"age der ersten Spalte: $2, -1, 0$. Vorzeichen der ersten Spalte (Schachbrett): $+, -, +$.
\begin{align*}
\det A &= (+)\,2 \cdot \det \begin{pmatrix} -4 & 1 \\ 2 & -5 \end{pmatrix} \;+\; (-)(-1) \cdot \det \begin{pmatrix} 5 & 3 \\ 2 & -5 \end{pmatrix} \;+\; (+)\,0 \cdot \dots \\
&= 2(20 - 2) + 1 \cdot (-25 - 6) + 0 \\
&= 36 - 31 + 0 = 5.
\end{align*}

$\det A = 5 \neq 0 \Rightarrow$ System eindeutig l\"osbar.

%% --- 30(a) Adjugate ---
\subsubsection*{(a) L\"osung \"uber die adjungierte Matrix}

\textbf{Plan:}
\begin{enumerate}[leftmargin=*, nosep]
    \item Alle 9 Kofaktoren $C_{ij}$ ausrechnen $\Rightarrow$ Kofaktormatrix.
    \item Kofaktormatrix transponieren $\Rightarrow$ Adjunkte $\text{adj}(A)$.
    \item $A^{-1} = \tfrac{1}{\det A} \cdot \text{adj}(A)$, dann $x = A^{-1} b$.
\end{enumerate}

\textbf{Schritt 2: Alle 9 Kofaktoren ausrechnen.}

Pro Kofaktor $C_{ij}$: in der Original-Matrix Zeile $i$ und Spalte $j$ grau wegstreichen $\Rightarrow$ verbleibende $2\times 2$-Matrix. Vorzeichen aus dem Schachbrettmuster $\bigl(\begin{smallmatrix} + & - & + \\ - & + & - \\ + & - & + \end{smallmatrix}\bigr)$.

\textbf{Zeile 1 der Kofaktormatrix:}

\[
C_{11}: \;
\begin{pmatrix}
\color{gray}2 & \color{gray}5 & \color{gray}3 \\
\color{gray}{-1} & -4 & 1 \\
\color{gray}0 & 2 & -5
\end{pmatrix}
\;\longrightarrow\;
+\det \begin{pmatrix} -4 & 1 \\ 2 & -5 \end{pmatrix} = 20 - 2 = 18.
\]

\[
C_{12}: \;
\begin{pmatrix}
\color{gray}2 & \color{gray}5 & \color{gray}3 \\
-1 & \color{gray}{-4} & 1 \\
0 & \color{gray}2 & -5
\end{pmatrix}
\;\longrightarrow\;
-\det \begin{pmatrix} -1 & 1 \\ 0 & -5 \end{pmatrix} = -(5 - 0) = -5.
\]

\[
C_{13}: \;
\begin{pmatrix}
\color{gray}2 & \color{gray}5 & \color{gray}3 \\
-1 & -4 & \color{gray}1 \\
0 & 2 & \color{gray}{-5}
\end{pmatrix}
\;\longrightarrow\;
+\det \begin{pmatrix} -1 & -4 \\ 0 & 2 \end{pmatrix} = -2 - 0 = -2.
\]

\textbf{Zeile 2 der Kofaktormatrix:}

\[
C_{21}: \;
\begin{pmatrix}
\color{gray}2 & 5 & 3 \\
\color{gray}{-1} & \color{gray}{-4} & \color{gray}1 \\
\color{gray}0 & 2 & -5
\end{pmatrix}
\;\longrightarrow\;
-\det \begin{pmatrix} 5 & 3 \\ 2 & -5 \end{pmatrix} = -(-25 - 6) = 31.
\]

\[
C_{22}: \;
\begin{pmatrix}
2 & \color{gray}5 & 3 \\
\color{gray}{-1} & \color{gray}{-4} & \color{gray}1 \\
0 & \color{gray}2 & -5
\end{pmatrix}
\;\longrightarrow\;
+\det \begin{pmatrix} 2 & 3 \\ 0 & -5 \end{pmatrix} = -10 - 0 = -10.
\]

\[
C_{23}: \;
\begin{pmatrix}
2 & 5 & \color{gray}3 \\
\color{gray}{-1} & \color{gray}{-4} & \color{gray}1 \\
0 & 2 & \color{gray}{-5}
\end{pmatrix}
\;\longrightarrow\;
-\det \begin{pmatrix} 2 & 5 \\ 0 & 2 \end{pmatrix} = -(4 - 0) = -4.
\]

\textbf{Zeile 3 der Kofaktormatrix:}

\[
C_{31}: \;
\begin{pmatrix}
\color{gray}2 & 5 & 3 \\
\color{gray}{-1} & -4 & 1 \\
\color{gray}0 & \color{gray}2 & \color{gray}{-5}
\end{pmatrix}
\;\longrightarrow\;
+\det \begin{pmatrix} 5 & 3 \\ -4 & 1 \end{pmatrix} = 5 + 12 = 17.
\]

\[
C_{32}: \;
\begin{pmatrix}
2 & \color{gray}5 & 3 \\
-1 & \color{gray}{-4} & 1 \\
\color{gray}0 & \color{gray}2 & \color{gray}{-5}
\end{pmatrix}
\;\longrightarrow\;
-\det \begin{pmatrix} 2 & 3 \\ -1 & 1 \end{pmatrix} = -(2 + 3) = -5.
\]

\[
C_{33}: \;
\begin{pmatrix}
2 & 5 & \color{gray}3 \\
-1 & -4 & \color{gray}1 \\
\color{gray}0 & \color{gray}2 & \color{gray}{-5}
\end{pmatrix}
\;\longrightarrow\;
+\det \begin{pmatrix} 2 & 5 \\ -1 & -4 \end{pmatrix} = -8 + 5 = -3.
\]

\textbf{Schritt 3: Kofaktormatrix bauen und transponieren.}

Alle $C_{ij}$ in die Position $(i, j)$ einer $3 \times 3$-Matrix eintragen, dann transponieren (Zeilen und Spalten tauschen):
\[
\text{cof}(A) = \begin{pmatrix} \mathbf{18} & -5 & -2 \\ 31 & \mathbf{-10} & -4 \\ 17 & -5 & \mathbf{-3} \end{pmatrix}
\;\xrightarrow{\text{transponieren}}\;
\text{adj}(A) = \begin{pmatrix} \mathbf{18} & 31 & 17 \\ -5 & \mathbf{-10} & -5 \\ -2 & -4 & \mathbf{-3} \end{pmatrix}.
\]
(Diagonale bleibt fett markiert; sie steht beim Transponieren still.)

\textbf{Schritt 4: $x = \tfrac{1}{\det A} \cdot \text{adj}(A) \cdot b$.}

\textbf{(a) Adjunkte mal $b$.} Pro Zeile von $\text{adj}(A)$ das Skalarprodukt mit $b = (5, -3, -1)^\top$:
\begin{itemize}[leftmargin=*, nosep]
    \item Zeile 1: $18 \cdot 5 + 31 \cdot (-3) + 17 \cdot (-1) = 90 - 93 - 17 = -20$.
    \item Zeile 2: $-5 \cdot 5 + (-10)(-3) + (-5)(-1) = -25 + 30 + 5 = 10$.
    \item Zeile 3: $-2 \cdot 5 + (-4)(-3) + (-3)(-1) = -10 + 12 + 3 = 5$.
\end{itemize}

\textbf{(b) Durch $\det A = 5$ teilen.}
\[
x = \frac{1}{5} \begin{pmatrix} -20 \\ 10 \\ 5 \end{pmatrix} = \begin{pmatrix} -4 \\ 2 \\ 1 \end{pmatrix}.
\]

\[
\fbox{\parbox{0.93\textwidth}{\centering
$x = (x_1, x_2, x_3) = (-4,\; 2,\; 1)$.
}}
\]

%% --- 30(b) Cramer ---
\subsubsection*{(b) L\"osung mit der Cramer'schen Regel}

\textbf{Idee.} F\"ur jede Variable $x_i$ ersetzt man in $A$ die $i$-te Spalte durch $b$ und berechnet die Determinante der so entstandenen Matrix $A_i$. Dann:
\[
x_i = \frac{\det A_i}{\det A}.
\]

\textbf{Schritt 5: Drei Hilfsmatrizen aufstellen.}

\textbf{Beispiel ausf\"uhrlich: $A_1$.} Spalte 1 von $A$ wird durch $b$ ersetzt:
\[
A = \begin{pmatrix} \color{gray}2 & 5 & 3 \\ \color{gray}{-1} & -4 & 1 \\ \color{gray}0 & 2 & -5 \end{pmatrix}
\;\xrightarrow{\text{Spalte 1} \leftarrow b}\;
A_1 = \begin{pmatrix} \color{red}5 & 5 & 3 \\ \color{red}{-3} & -4 & 1 \\ \color{red}{-1} & 2 & -5 \end{pmatrix}
\]
\[
\det A_1 = 5(20 - 2) - 5(15 + 1) + 3(-6 - 4) = 90 - 80 - 30 = -20.
\]

\textbf{Analog $A_2$ (Spalte 2 $\leftarrow b$) und $A_3$ (Spalte 3 $\leftarrow b$):}
\[
A_2 = \begin{pmatrix} 2 & \color{red}5 & 3 \\ -1 & \color{red}{-3} & 1 \\ 0 & \color{red}{-1} & -5 \end{pmatrix},\;
\det A_2 = 2(15 + 1) - 5(5 - 0) + 3(1 - 0) = 10.
\]
\[
A_3 = \begin{pmatrix} 2 & 5 & \color{red}5 \\ -1 & -4 & \color{red}{-3} \\ 0 & 2 & \color{red}{-1} \end{pmatrix},\;
\det A_3 = 2(4 + 6) - 5(1 - 0) + 5(-2 - 0) = 5.
\]

\textbf{Schritt 6: $x_i = \det A_i / \det A$.}

\[
x_1 = \frac{\det A_1}{\det A} = \frac{-20}{5} = -4, \quad
x_2 = \frac{\det A_2}{\det A} = \frac{10}{5} = 2, \quad
x_3 = \frac{\det A_3}{\det A} = \frac{5}{5} = 1.
\]

\[
\fbox{\parbox{0.93\textwidth}{\centering
$x = (-4,\; 2,\; 1)$ -- gleich wie in (a). $\checkmark$
}}
\]

\textbf{Probe:}
\begin{align*}
2(-4) + 5(2) + 3(1) &= -8 + 10 + 3 = 5 \;\checkmark \\
-(-4) - 4(2) + 1 &= 4 - 8 + 1 = -3 \;\checkmark \\
2(2) - 5(1) &= 4 - 5 = -1 \;\checkmark
\end{align*}
